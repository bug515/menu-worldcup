        <textarea id="menu-input" placeholder="메뉴를 한 줄에 하나씩 입력..."></textarea>

        <button onclick="initGame()">월드컵 시작하기</button>

<script>

    let items = [];

    let winners = [];

    let currentIndex = 0;

    let isProcessing = false;

    // 배열 무작위 섞기 (Fisher-Yates)

    function shuffle(array) {

        for (let i = array.length - 1; i > 0; i--) {

            const j = Math.floor(Math.random() * (i + 1));

            [array[i], array[j]] = [array[j], array[i]];

        }

    }

    function initGame() {

        const text = document.getElementById('menu-input').value.trim();

        if (!text) { alert("메뉴를 입력해주세요!"); return; }

        items = text.split('\n').map(s => s.trim()).filter(s => s.length > 0);

        if (items.length < 2) { alert("최소 2개 이상의 메뉴가 필요합니다."); return; }

        document.getElementById('setup-area').style.display = 'none';

        document.getElementById('result-area').style.display = 'none';

        document.getElementById('game-area').style.display = 'block';

       

        winners = [];

        startNewRound();

    }

    function startNewRound() {

        currentIndex = 0;

        const byeMsgArea = document.getElementById('bye-msg');

        byeMsgArea.innerText = "";

        // 1. 라운드 시작 전 리스트 전체 셔플

        shuffle(items);

        // 2. 최종 승자 결정

        if (items.length === 1) {

            showResult(items[0]);

            return;

        }

        // 3. 홀수일 때 부전승 무작위 선택

        if (items.length % 2 !== 0) {

            // 셔플된 상태이므로 맨 마지막 항목을 빼서 부전승 처리

            const byeItem = items.pop();

            winners.push(byeItem);

            byeMsgArea.innerText = `✨ 이번 라운드 부전승: ${byeItem}`;

        }

        document.getElementById('round-title').innerText = `${items.length + (items.length % 2 === 0 ? 0 : 1)}강 진행 중`;

        updateMatchUI();

    }

    function updateMatchUI() {

        if (currentIndex >= items.length) {

            items = [...winners];

            winners = [];

            startNewRound();

            return;

        }

        document.getElementById('choice1').innerText = items[currentIndex];

        document.getElementById('choice2').innerText = items[currentIndex + 1];

        document.getElementById('choice1').classList.remove('active-select');

        document.getElementById('choice2').classList.remove('active-select');

        document.getElementById('match-count').innerText = `현재 경기: ${Math.floor(currentIndex/2) + 1} / ${Math.floor(items.length/2)}`;

        isProcessing = false;

    }

    function handleSelect(n, element) {

        if (isProcessing) return;

        isProcessing = true;

        element.classList.add('active-select');

        setTimeout(() => {

            winners.push(items[currentIndex + n]);

            currentIndex += 2;

            updateMatchUI();

        }, 500); // 0.5초 대기

    }

    function showResult(winner) {

        document.getElementById('game-area').style.display = 'none';

        document.getElementById('result-area').style.display = 'block';

        document.getElementById('result-text').innerText = winner;

    }

    function restartGame() {

        document.getElementById('result-area').style.display = 'none';

        document.getElementById('setup-area').style.display = 'block';

    }

</script>
